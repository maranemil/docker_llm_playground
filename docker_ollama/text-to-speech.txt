Using sed in a Pipeline
echo '{"name": "Alice\\nBaker", "age": 30}' | \
jq '.name' | \
sed 's/\\n//g'   
Output: "AliceBaker"

Remove Character Before Parsing
echo '{"name": "Alice\\nBaker"}' | \
sed 's/\\n//g' | \
jq '.name'
Output: "AliceBaker"

Using tr to Delete Characters
echo '{"msg": "test\\tdata"}' | \
jq -r '.msg' | \
tr -d '\t'
Output: testdata

Handle Raw Output with jq -r
echo '{"text": "hello\\tworld"}' | \
jq -r '.text' | \
sed 's/\\t/ /g'
Output: hello world

For control characters like \u0000, use:
jq '.data' file.json | sed 's/\\u0000//g'

Handle Large Audio Files
Split long audio files into chunks and transcribe them efficiently.

import speech_recognition as sr
from pydub import AudioSegment
from pydub.silence import split_on_silence
import os

def transcribe_large_audio(path):
    sound = AudioSegment.from_wav(path)
    chunks = split_on_silence(
        sound,
        min_silence_len=500,
        silence_thresh=sound.dBFS - 14,
        keep_silence=500
    )

    folder_name = "audio-chunks"
    if not os.path.exists(folder_name):
        os.mkdir(folder_name)

    recognizer = sr.Recognizer()
    full_text = ""

    for i, chunk in enumerate(chunks):
        chunk_filename = f"{folder_name}/chunk{i}.wav"
        chunk.export(chunk_filename, format="wav")

        with sr.AudioFile(chunk_filename) as source:
            audio = recognizer.record(source)
            try:
                text = recognizer.recognize_google(audio)
                full_text += text + " "
            except sr.UnknownValueError:
                print(f"Chunk {i}: Could not understand.")
            except sr.RequestError:
                print(f"Chunk {i}: API connection error.")

    return full_text

# Usage
print("Full transcript:", transcribe_large_audio("long_audio.wav"))

Offline Speech Recognition (Optional)


# pip install vosk pyaudio
from vosk import Model, KaldiRecognizer
import pyaudio

model = Model("vosk-model-small-en-us")  # Download from https://alphacephei.com/vosk/models
rec = KaldiRecognizer(model, 16000)

p = pyaudio.PyAudio()
stream = p.open(format=pyaudio.paInt16, channels=1, rate=16000, input=True, frames_per_buffer=8192)
stream.start_stream()

print("Listening (say 'exit' to stop)...")
while True:
    data = stream.read(4096)
    if rec.AcceptWaveform(data):
        result = rec.Result()
        print(result)
        if "exit" in result.lower():
            break
