// Import required modules
const express = require('express');
const bodyParser = require('body-parser');

const app = express();
const PORT = 3000;

// Middleware to parse JSON request bodies
app.use(bodyParser.json());

// In-memory "database"
let items = [
    { id: 1, name: "Item One" },
    { id: 2, name: "Item Two" }
];

// ✅ CREATE - Add a new item
app.post('/items', (req, res) => {
    try {
        const { name } = req.body;
        if (!name || typeof name !== 'string') {
            return res.status(400).json({ error: "Invalid 'name' field" });
        }
        const newItem = { id: items.length ? items[items.length - 1].id + 1 : 1, name };
        items.push(newItem);
        res.status(201).json(newItem);
    } catch (err) {
        res.status(500).json({ error: "Server error" });
    }
});

// 📖 READ - Get all items
app.get('/items', (req, res) => {
    res.json(items);
});

// 📖 READ - Get single item by ID
app.get('/items/:id', (req, res) => {
    const id = parseInt(req.params.id);
    const item = items.find(i => i.id === id);
    if (!item) {
        return res.status(404).json({ error: "Item not found" });
    }
    res.json(item);
});

// ✏️ UPDATE - Modify an item
app.put('/items/:id', (req, res) => {
    const id = parseInt(req.params.id);
    const { name } = req.body;
    if (!name || typeof name !== 'string') {
        return res.status(400).json({ error: "Invalid 'name' field" });
    }
    const itemIndex = items.findIndex(i => i.id === id);
    if (itemIndex === -1) {
        return res.status(404).json({ error: "Item not found" });
    }
    items[itemIndex].name = name;
    res.json(items[itemIndex]);
});

// 🗑 DELETE - Remove an item
app.delete('/items/:id', (req, res) => {
    const id = parseInt(req.params.id);
    const itemIndex = items.findIndex(i => i.id === id);
    if (itemIndex === -1) {
        return res.status(404).json({ error: "Item not found" });
    }
    const deletedItem = items.splice(itemIndex, 1);
    res.json(deletedItem[0]);
});

// Start the server
app.listen(PORT, () => {
    console.log(`✅ Server running on http://localhost:${PORT}`);
});

Creating a REST API with Node.js and Express is a common approach for building scalable backend services. Below is a concise guide to get you started.

1. Initialize the Project

Open a terminal and run:

mkdir my-api
cd my-api
npm init -y
Copy
Install dependencies:

npm install express mongoose body-parser cors
Copy
Here, Express handles routing, Mongoose connects to MongoDB, body-parser parses JSON, and CORS enables cross-origin requests.

2. Set Up the Server

Create server.js:

const express = require('express');
const mongoose = require('mongoose');
const bodyParser = require('body-parser');
const cors = require('cors');

const app = express();
const PORT = process.env.PORT || 5000;

app.use(cors());
app.use(bodyParser.json());

mongoose.connect('your_mongodb_connection_string', {
useNewUrlParser: true,
useUnifiedTopology: true
}).then(() => console.log('MongoDB connected'))
.catch(err => console.log(err));

app.get('/status', (req, res) => {
res.json({ status: 'Running', time: new Date().toISOString() });
});

app.listen(PORT, () => console.log(`Server running on port ${PORT}`));

4. Define CRUD Routes

In routes/items.js:

const express = require('express');
const router = express.Router();
const Item = require('../models/Item');

// CREATE
router.post('/items', async (req, res) => {
try {
const item = await Item.create(req.body);
res.status(201).json(item);
} catch (err) {
res.status(500).json({ error: err.message });
}
});

// READ
router.get('/items', async (req, res) => {
const items = await Item.find();
res.json(items);
});

// UPDATE
router.put('/items/:id', async (req, res) => {
const updated = await Item.findByIdAndUpdate(req.params.id, req.body, { new: true });
res.json(updated);
});

// DELETE
router.delete('/items/:id', async (req, res) => {
await Item.findByIdAndDelete(req.params.id);
res.status(204).send();
});

module.exports = router;
