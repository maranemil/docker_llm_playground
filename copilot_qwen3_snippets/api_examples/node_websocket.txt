It includes connection handling, message broadcasting, and error handling.

2️⃣ server.js — WebSocket API
Javascript

Copy code
// Import required modules
const WebSocket = require('ws');
const http = require('http');

// Create an HTTP server (optional, for future REST + WS integration)
const server = http.createServer();

// Create a WebSocket server attached to the HTTP server
const wss = new WebSocket.Server({ server });

// Broadcast helper function
function broadcast(message, sender) {
    wss.clients.forEach(client => {
        if (client !== sender && client.readyState === WebSocket.OPEN) {
            client.send(message);
        }
    });
}

// Handle new WebSocket connections
wss.on('connection', (ws) => {
    console.log('New client connected');

    // Send welcome message to the new client
    ws.send(JSON.stringify({ type: 'info', message: 'Welcome to WebSocket API!' }));

    // Handle incoming messages
    ws.on('message', (data) => {
        try {
            const msg = data.toString();
            console.log(`Received: ${msg}`);
            // Broadcast to all other clients
            broadcast(msg, ws);
        } catch (err) {
            console.error('Error processing message:', err);
            ws.send(JSON.stringify({ type: 'error', message: 'Invalid message format' }));
        }
    });

    // Handle client disconnection
    ws.on('close', () => {
        console.log('Client disconnected');
    });

    // Handle errors
    ws.on('error', (err) => {
        console.error('WebSocket error:', err);
    });
});

// Start the server
const PORT = 8080;
server.listen(PORT, () => {
    console.log(`WebSocket server running on ws://localhost:${PORT}`);
});

WebSockets enable persistent, full-duplex communication between a client and a server over a single TCP connection. Unlike HTTP’s request-response model, WebSockets allow real-time, bidirectional data exchange with minimal overhead, making them ideal for chat apps, live dashboards, multiplayer games, and IoT systems.

From Node.js v21+, a native WebSocket client is available via the WebSocket constructor, removing the need for external client libraries like ws for outbound connections. However, Node.js still lacks a built-in WebSocket server, so libraries like ws or Socket.IO are required for hosting WebSocket endpoints.

Example: Native WebSocket Client (Node.js v22+)

// Native WebSocket client
const socket = new WebSocket('ws://localhost:8080');

socket.addEventListener('open', () => {
console.log('Connected to server');
socket.send(JSON.stringify({ type: 'greet', message: 'Hello Server!' }));
});

socket.addEventListener('message', (event) => {
console.log('Received:', event.data);
});

socket.addEventListener('close', (event) => {
console.log('Connection closed:', event.code, event.reason);
});

socket.addEventListener('error', (err) => {
console.error('WebSocket error:', err);
});

This approach is standards-compliant and interoperable with browser WebSockets.

Example: WebSocket Server with ws

import { WebSocketServer } from 'ws';

const wss = new WebSocketServer({ port: 8080 });
console.log('Server running on ws://localhost:8080');

wss.on('connection', (ws) => {
console.log('Client connected');
ws.send('Welcome!');

ws.on('message', (msg) => {
console.log('Received:', msg.toString());
ws.send(`Echo: ${msg}`);
});

ws.on('close', () => console.log('Client disconnected'));
});
Copy
The ws library is fast, lightweight, and supports features like per-message compression, binary data, and connection heartbeats.

Best Practices

Authentication: Validate tokens during the HTTP upgrade request.

Heartbeat/Ping: Detect broken connections with periodic pings.

Rate Limiting: Prevent abuse by limiting messages per connection.

Compression: Use perMessageDeflate for bandwidth optimization.

Key Takeaway: Use native WebSocket client in modern Node.js for outbound connections, and ws or similar libraries for server-side implementations. Always secure, validate, and monitor connections for production-grade real-time apps
