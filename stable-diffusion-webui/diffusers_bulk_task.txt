from diffusers import StableDiffusionPipeline
import torch
import time
timestr = time.strftime("%Y%m%d-%H%M%S")

import sys

# sys.argv[0] is the script name
# sys.argv[1] is the first argument
print("Script:", sys.argv[0])
print("Arg 1:", sys.argv[1] if len(sys.argv) > 1 else "None")

promptstyle = ""
if len(sys.argv) > 1:
    promptstyle = sys.argv[1]
if len(sys.argv) > 2:
    promptstyle += "_" + sys.argv[2]
print(promptstyle)

model_id = "dreamlike-art/dreamlike-photoreal-2.0"
pipe = StableDiffusionPipeline.from_pretrained(model_id, torch_dtype=torch.float32)
pipe = pipe.to("cpu")

prompt = promptstyle + " artstyle,a photo of an astronaut, hafltone, anti alias, sharp,"
#prompt = "photo, a church in the middle of a field of crops, bright cinematic lighting, gopro, fisheye lens"
#image = pipe(prompt).images[0]
image = pipe(
    prompt= promptstyle + " artstyle,a photo of an astronaut, hafltone, anti alias, sharp,",
    num_inference_steps=13,       # Recommended: 25-30 for DPMSolver++
    guidance_scale=7.5,           # Controls prompt adherence
    height=456,   # Default for SD 1.x 512
    width=312,    # Landscape ratio  768

#    clip_skip=2,
).images[0]
print(prompt)

image.save(timestr + "_" + promptstyle + "_dreamlike.png")
